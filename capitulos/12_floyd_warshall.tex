\chapter{Caminhos mínimos entre todos os pares: Floyd-Warshall}\label{cap:fw}\label{sec:fw}
% ================================================================
\begin{objetivos}
\begin{itemize}
  \item Notação e representações de grafos usadas no restante do livro.
  \item O problema dos caminhos mínimos entre todos os pares de vértices.
  \item O algoritmo de Floyd-Warshall como programação dinâmica, com reconstrução dos caminhos.
  \item Pesos negativos, ciclos negativos e a comparação com o algoritmo de Dijkstra.
\end{itemize}
\end{objetivos}

O algoritmo de Floyd-Warshall é mais uma aplicação da programação dinâmica, agora sobre um grafo: queremos o custo do caminho mais barato entre \emph{todos} os pares de vértices. Por ser o primeiro algoritmo em grafos do livro, o capítulo começa fixando a notação e as representações de grafos usadas daqui em diante.

\section{Grafos: notação e representação}\label{sec:buscas}
Um grafo é um par $G=(V,E)$, em que $V$ é o conjunto de vértices e $E$ o de arestas; escrevemos $n=|V|$ e $m=|E|$. A \textbf{vizinhança}\index{vizinhanca@vizinhança} de um vértice $v$, $N(v)$, é o conjunto dos vértices ligados a $v$ por uma aresta, e o seu \textbf{grau} é $|N(v)|$. Num \textbf{digrafo}\index{digrafo} (grafo direcionado), cada aresta $v\to w$ tem uma direção, e $N_{out}(v)$ é o conjunto dos vértices $w$ com uma aresta $v\to w$. Há duas formas usuais de representar um grafo na memória:
\begin{center}\small
\begin{tabular}{@{}lccc@{}}
\toprule
Representação & Memória & ``$vw\in E$?'' & Percorrer $N(v)$\\
\midrule
Matriz de adjacência (ou de pesos $W$) & $\Theta(n^2)$ & $O(1)$ & $\Theta(n)$\\
Listas de adjacência & $\Theta(n+m)$ & $O(\text{grau}(v))$ & $\Theta(\text{grau}(v))$\\
\bottomrule
\end{tabular}
\end{center}

\begin{exemplo}[As duas representações]\label{ex:repgrafo}
O grafo da Figura~\ref{fig:repgrafo} tem $n=4$ vértices e $m=4$ arestas: $12$, $13$, $23$ e $34$. Na matriz de adjacência, a posição $(v,w)$ vale 1 se $vw$ é aresta; como o grafo não é direcionado, a matriz é simétrica, e cada aresta aparece duas vezes. Nas listas de adjacência, cada vértice guarda só os seus vizinhos: $N(3)=\{1,2,4\}$, e o grau de 3 é 3. Para saber se $24$ é aresta, a matriz responde com uma consulta, à posição $(2,4)$; as listas precisam percorrer $N(2)=\{1,3\}$. Em compensação, as listas ocupam espaço proporcional a $n+2m=12$ entradas, contra $n^2=16$ da matriz --- uma diferença que, num grafo esparso com um milhão de vértices, separa alguns megabytes de terabytes.

A matriz é adequada a grafos densos, com $m$ próximo de $n^2$, e a testes frequentes de adjacência; as listas, a grafos esparsos e a algoritmos que percorrem vizinhanças. Uma \textbf{busca}, que visita todos os vértices e arestas do grafo examinando cada vizinhança uma vez, custa $\Theta(n+m)$ com listas e $\Theta(n^2)$ com matriz; as buscas em largura e em profundidade são estudadas nos Capítulos~\ref{cap:bfs} e~\ref{cap:dfs}.

\section{Matriz de pesos}\label{sec:fwmatriz}
Em problemas de caminhos mínimos, cada aresta tem um peso (um custo, uma distância, um tempo), e o grafo é dado por uma \textbf{matriz de pesos}\index{matriz de pesos} $W$: $W(x,x)=0$; $W(x,y)$ é o peso da aresta $xy$, se ela existe; e $W(x,y)=\infty$ se não existe. O custo de um caminho é a soma dos pesos das suas arestas, e a \textbf{distância}\index{distancia@distância} $\operatorname{dist}(x,y)$ é o menor custo de um caminho de $x$ a $y$. Um grafo não direcionado é tratado como um digrafo em que cada aresta $xy$ de peso $p$ dá origem às arestas $x\to y$ e $y\to x$, ambas de peso $p$.

Usaremos como exemplo, neste capítulo e no Capítulo~\ref{cap:dijkstra}, o digrafo de cinco vértices dado pela matriz abaixo e desenhado na Figura~\ref{fig:digrafofw}; os vértices são chamados $1,\dots,5$ aqui e $a,\dots,e$ no capítulo do algoritmo de Dijkstra.

1\,($a$) & 0 & 1 & 3 & $\infty$ & 6\\

4\,($d$) & 3 & $\infty$ & $\infty$ & 0 & 2\\

\section{O algoritmo de Floyd-Warshall}\label{sec:fwalg}\index{floyd-warshall, algoritmo de@Floyd-Warshall, algoritmo de}
\textbf{Problema:} dado um digrafo com vértices $1,\dots,n$ e matriz de pesos $W$, calcular $\operatorname{dist}(i,j)$ para todos os pares ordenados $(i,j)$. A ideia é limitar, passo a passo, quais vértices podem aparecer \emph{no meio} dos caminhos:
\[
D^k(i,j)=\text{custo do caminho mínimo de }i\text{ a }j\text{ cujos vértices \emph{intermediários} estão em }\{1,\dots,k\}.
\]
Então $D^0=W$, pois sem intermediários só valem as arestas diretas, e $D^n$ é a matriz das distâncias.

\begin{intuicao}
Imagine que as cidades $1,\dots,n$ abrem, uma por vez, para servir de conexão. No começo, só há voos diretos ($D^0=W$). Quando a cidade 1 abre, alguns trajetos ficam mais baratos fazendo escala nela; quando a 2 abre, trajetos com escala em 2 --- e possivelmente também em 1 --- entram em jogo; e assim por diante. Cada abertura é testada para todos os pares de uma vez, e depois que todas as cidades abriram, todos os trajetos possíveis foram considerados.
\end{intuicao}

\textbf{Decomposição.} Um caminho mínimo de $i$ a $j$ com intermediários em $\{1,\dots,k\}$ ou \textbf{não passa} por $k$, e custa $D^{k-1}(i,j)$, ou passa \textbf{exatamente uma vez} por $k$ --- voltar a $k$ formaria um ciclo, que não ajuda --- e se divide num caminho mínimo de $i$ a $k$ e num de $k$ a $j$, ambos com intermediários em $\{1,\dots,k-1\}$ (Figura~\ref{fig:fwdecomp}):
\[
D^k(i,j)=\begin{cases}W(i,j) & k=0\\ \min\{\,D^{k-1}(i,j),\ D^{k-1}(i,k)+D^{k-1}(k,j)\,\} & k\ge1.\end{cases}
\]

\begin{figura}[As duas possibilidades para um caminho mínimo de $i$ a $j$ com intermediários em $\{1,\dots,k\}$: ou ele não usa $k$ (em cima), ou passa por $k$ uma única vez, e os dois trechos só usam intermediários em $\{1,\dots,k-1\}$ (embaixo).\label{fig:fwdecomp}]
\begin{tikzpicture}[x=1cm,y=1cm]

\draw[seta, gray!70!black, densely dashed] (i) to[bend left=20] node[above, rot] {intermediários em $\{1,\dots,k-1\}$: custo $D^{k-1}(i,j)$} (j);

\draw[seta, MidnightBlue, densely dashed] (k) -- node[below right, rot] {$D^{k-1}(k,j)$} (j);
\end{tikzpicture}
\end{figura}

\textbf{Por que só ``prefixos'' $\{1,\dots,k\}$?} Uma formulação mais livre usaria, para cada conjunto $S\subseteq V$ de intermediários permitidos, $D(i,j,S)=\min_{v\in S}\{D(i,v,S\setminus\{v\})+D(v,j,S\setminus\{v\})\}$. Ela também é correta, mas tem $n^2\,2^n$ subproblemas. Fixando uma ordem dos vértices e usando só os $n+1$ prefixos, sobram $O(n^2)$ subproblemas por valor de $k$, e $O(n^3)$ no total. Com $n=20$, por exemplo, são $20^2\cdot2^{20}\approx4{,}2\cdot10^8$ subproblemas na formulação por subconjuntos, contra $20^3=8000$ passos do Floyd-Warshall.

\begin{pseudo}{Floyd-Warshall com matriz de predecessores $P$: $\Theta(n^3)$ de tempo e $\Theta(n^2)$ de memória}
\begin{algorithmic}
\Function{FloydWarshall}{$W,n$}
  \For{$i,j:=1$ \textbf{até} $n$}
    \State $D[i,j]:=W[i,j]$
    \If{$i\ne j$ \textbf{e} $W[i,j]<\infty$} \State $P[i,j]:=i$ \Else \State $P[i,j]:=\nil$ \EndIf
  \EndFor
  \For{$k:=1$ \textbf{até} $n$} \Comment{$k$ é o laço \textbf{externo}}
    \For{$i:=1$ \textbf{até} $n$}
      \For{$j:=1$ \textbf{até} $n$}
        \If{$D[i,k]+D[k,j]<D[i,j]$}
          \State $D[i,j]:=D[i,k]+D[k,j]$;\ $P[i,j]:=P[k,j]$ \Comment{penúltimo vértice do trecho $k\to j$}
        \EndIf
      \EndFor
    \EndFor
  \EndFor
  \State \Return $D,P$
\EndFunction
\Procedure{Caminho}{$P,i,j$}
  \If{$i=j$} \State imprima $i$
  \ElsIf{$P[i,j]=\nil$} \State imprima ``não há caminho''
  \Else \State \Call{Caminho}{$P,i,P[i,j]$};\ imprima $j$
  \EndIf
\EndProcedure
\end{algorithmic}
\end{pseudo}
A matriz $P$ guarda, em $P^k(i,j)$, o \textbf{penúltimo} vértice do caminho mínimo de $i$ a $j$ com intermediários em $\{1,\dots,k\}$. Se o caminho não melhora ao permitir $k$, $P^k(i,j)=P^{k-1}(i,j)$; se melhora, o novo caminho termina com o trecho de $k$ a $j$, e $P^k(i,j)=P^{k-1}(k,j)$. Uma única matriz $D$ basta, com atualizações ``no lugar'': na iteração $k$, a linha e a coluna $k$ não mudam, pois $D(k,k)=0$.

\begin{exemplo}[A primeira iteração, $k=1$]\label{ex:fwk1}
Na iteração $k=1$, cada entrada $D(i,j)$ é comparada com $D(i,1)+D(1,j)$: o caminho de $i$ a $j$ com escala em 1. Só interessam os $i$ que chegam a 1 (os finitos da coluna 1: $i=3$, com custo 1, e $i=4$, com custo 3) e os $j$ alcançados a partir de 1 (os finitos da linha 1: $j=2$, 3 e 5). As candidatas são, portanto, $D(3,2)$, $D(3,5)$, $D(4,2)$, $D(4,3)$ e $D(4,5)$ (Figura~\ref{fig:fwk1}):
\begin{itemize}
  \item $D(3,5)=\min\{\infty,\ D(3,1)+D(1,5)\}=\min\{\infty,\,1+6\}=7$: o caminho $3\to1\to5$;
  \item $D(4,2)=\min\{\infty,\,3+1\}=4$ e $D(4,3)=\min\{\infty,\,3+3\}=6$;
  \item $D(3,2)=\min\{2,\,1+1\}=2$ e $D(4,5)=\min\{2,\,3+6\}=2$ não mudam: o caminho direto já é tão bom quanto o que passa por 1.
\end{itemize}

\begin{itemize}
  \item \textbf{Pesos negativos.} O algoritmo funciona com arestas de peso negativo desde que \textbf{não haja ciclo negativo}, pois então o caminho mínimo nunca repete vértices. Há um ciclo negativo se e somente se, ao final, algum $D(i,i)$ é negativo.
  \item \textbf{Floyd-Warshall ou $n$ execuções do Dijkstra?} O Floyd-Warshall custa $\Theta(n^3)$ e é muito simples. Executar o algoritmo de Dijkstra (Capítulo~\ref{cap:dijkstra}) a partir de cada vértice custa $O(n\,(n+m)\log n)$ com heap binário, ou $O(n^3)$ com vetor. Em grafos esparsos e sem pesos negativos, as $n$ execuções do Dijkstra são mais rápidas; em grafos densos, ou com pesos negativos, o Floyd-Warshall é preferível.
\end{itemize}

\begin{exemplo}[Pesos negativos, com e sem ciclo negativo]\label{ex:fwneg}
No digrafo com as arestas $1\to2$ (4), $2\to3$ ($-2$) e $1\to3$ (3), a aresta direta de 1 a 3 custa 3, mas o caminho $1\to2\to3$ custa $4-2=2$. O algoritmo encontra isso na iteração $k=2$: $D(1,3)=\min\{3,\,D(1,2)+D(2,3)\}=\min\{3,2\}=2$. Não há ciclos, e a diagonal termina toda nula.

Já no digrafo com as arestas $1\to2$ (1), $2\to3$ ($-4$) e $3\to1$ (2), o ciclo $1\to2\to3\to1$ pesa $1-4+2=-1$: dando voltas nele, o custo diminui sem limite, e ``caminho mínimo'' deixa de fazer sentido. Ao final do algoritmo, os três elementos da diagonal são negativos (por exemplo, $D(1,1)=-1$, o peso do ciclo), e é por esse sinal que o ciclo negativo é detectado.
\end{exemplo}

\begin{cuidado}[Armadilhas do Floyd-Warshall]
\begin{itemize}
  \item Colocar o laço de $k$ \textbf{por dentro} ($i$ e $j$ por fora) dá resposta errada: $k$ precisa ser o laço externo, pois $D^k$ só pode ser calculada depois de toda a $D^{k-1}$.
  \item $D^k(i,j)$ restringe os vértices \emph{intermediários} a $\{1,\dots,k\}$; os extremos $i$ e $j$ podem ser quaisquer.
  \item $P(i,j)$ é o \textbf{penúltimo} vértice do caminho, e não o segundo; a reconstrução vai de trás para frente.
  \item Com ciclo negativo, ``caminho mínimo'' deixa de fazer sentido; o algoritmo detecta a situação por $D(i,i)<0$.
\end{itemize}
\end{cuidado}

\begin{chave}
$D^k(i,j)=\min\{D^{k-1}(i,j),\ D^{k-1}(i,k)+D^{k-1}(k,j)\}$, com $D^0=W$; $k$ é o laço externo; $\Theta(n^3)$ de tempo e $\Theta(n^2)$ de memória, com uma única matriz. Quando o caminho por $k$ melhora, $P(i,j)$ recebe $P(k,j)$. Usar só os prefixos $\{1,\dots,k\}$, e não subconjuntos quaisquer, é o que torna o algoritmo polinomial.
\end{chave}
